## 3. VIEWS (2)

### `payment_analytics`
```sql
CREATE VIEW payment_analytics AS 
SELECT l.accountcode, l.transaction_amount, f.faculty_name, l.transactionDate
FROM campus_dynamics_accounts.fin_ledger l 
JOIN campus_dynamics.acad_faculty f
WHERE l.account_type LIKE '%Fees' 
  AND l.transactionType = 'CR' 
  AND l.transaction_amount > 0 
  AND REPLACE(l.account_type, 'Fees', '') = f.abbrev;
```

### `v_all_accounts`
```sql
CREATE VIEW v_all_accounts AS 
SELECT s.AccountCode AS account_code, s.AccountName AS account_name,
       s.MainAccountCode AS main_account_code, s.accounttype AS account_type,
       s.collectionLedgerType AS ledger_type, 'CHART' AS account_source
FROM campus_dynamics_accounts.fin_subaccounts s
UNION ALL
SELECT TRIM(st.regno) AS account_code,
       TRIM(CONCAT(IFNULL(st.firstname,''), 
         CASE WHEN TRIM(IFNULL(st.othername,'')) <> '' 
              THEN CONCAT(' ', TRIM(st.othername)) ELSE '' END)) AS account_name,
       'STUDENT' AS main_account_code,
       'Student Account' AS account_type,
       IFNULL(p.progname,'Unknown Programme') AS ledger_type,
       'STUDENT' AS account_source
FROM campus_dynamics.acad_student st 
LEFT JOIN campus_dynamics.acad_programme p ON TRIM(st.progid) = TRIM(p.progcode)
WHERE TRIM(IFNULL(st.regno,'')) <> '';
```
